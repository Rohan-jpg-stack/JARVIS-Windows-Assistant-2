def load_memory():

    global conversation

    if not os.path.exists(MEMORY_FILE):
        return

    try:

        with open(MEMORY_FILE, "r", encoding="utf-8") as file:
            data = json.load(file)

        if isinstance(data, list) and data:
            conversation = data

        logging.info("Memory loaded.")

    except Exception as e:

        logging.exception(e)

        conversation = [
            {
                "role": "system",
                "content": SYSTEM_PROMPT
            }
        ]


def save_memory():

    try:

        with open(MEMORY_FILE, "w", encoding="utf-8") as file:
            json.dump(
                conversation,
                file,
                indent=4,
                ensure_ascii=False
            )

    except Exception as e:
        logging.exception(e)


def clear_memory():

    global conversation

    conversation = [
        {
            "role": "system",
            "content": SYSTEM_PROMPT
        }
    ]

# SAFE CALCULATOR
# ============================================================

OPERATORS = {
    ast.Add: operator.add,
    ast.Sub: operator.sub,
    ast.Mult: operator.mul,
    ast.Div: operator.truediv,
    ast.Pow: operator.pow,
    ast.Mod: operator.mod,
    ast.USub: operator.neg,
    ast.UAdd: operator.pos,
}


def evaluate(node):

    if isinstance(node, ast.Constant):

        if isinstance(node.value, (int, float)):
            return node.value

        raise ValueError("Invalid value")

    if isinstance(node, ast.BinOp):

        operation = OPERATORS.get(type(node.op))

        if operation is None:
            raise ValueError("Operator not allowed")

        return operation(
            evaluate(node.left),
            evaluate(node.right)
        )

    if isinstance(node, ast.UnaryOp):

        operation = OPERATORS.get(type(node.op))

        if operation is None:
            raise ValueError("Operator not allowed")

        return operation(
            evaluate(node.operand)
        )

    raise ValueError("Invalid calculation")


def calculate(expression):

    tree = ast.parse(
        expression,
        mode="eval"
    )

    return evaluate(tree.body)

# TEXT TO SPEECH
# ============================================================

speech_lock = threading.Lock()


def speak(text):

    text = str(text)

    if not text.strip():
        return

    log_message("JARVIS", text)

    if not TTS_AVAILABLE:
        return

    def speak_thread():

        with speech_lock:

            try:
                engine.say(text)
                engine.runAndWait()

            except Exception as e:
                logging.exception(e)

    threading.Thread(
        target=speak_thread,
        daemon=True
    ).start()

def ask_ai(question):

    global conversation

    conversation.append(
        {
            "role": "user",
            "content": question
        }
    )

    try:

        response = ollama.chat(
            model=MODEL,
            messages=conversation
        )

        answer = response["message"]["content"].strip()

        conversation.append(
            {
                "role": "assistant",
                "content": answer
            }
        )

        save_memory()

        return answer

    except Exception as e:

        logging.exception(e)

        # Remove failed user message so memory does not
        # repeatedly store unanswered requests.
        if conversation and conversation[-1]["role"] == "user":
            conversation.pop()

        return (
            "I cannot connect to Ollama right now, Sir. "
            "Please make sure Ollama is running."
        )

def battery_status():

    battery = psutil.sensors_battery()

    if battery is None:
        return "Battery information is not available, Sir."

    percentage = battery.percent

    if battery.power_plugged:
        state = "and the computer is connected to power"
    else:
        state = "and the computer is running on battery"

    return (
        f"Battery is at {percentage:.0f} percent, "
        f"{state}, Sir."
    )


def system_status():

    cpu = psutil.cpu_percent(interval=1)
    ram = psutil.virtual_memory()

    ram_used = ram.percent

    return (
        f"CPU usage is {cpu:.0f} percent. "
        f"Memory usage is {ram_used:.0f} percent, Sir."
    )

def save_note(text):

    try:

        with open(
            NOTES_FILE,
            "a",
            encoding="utf-8"
        ) as file:

            timestamp = datetime.datetime.now().strftime(
                "%Y-%m-%d %H:%M:%S"
            )

            file.write(
                f"[{timestamp}] {text}\n"
            )

        return "Note saved, Sir."

    except Exception as e:

        logging.exception(e)

        return "I could not save the note, Sir."


def read_notes():

    if not os.path.exists(NOTES_FILE):
        return "You do not have any saved notes, Sir."

    try:

        with open(
            NOTES_FILE,
            "r",
            encoding="utf-8"
        ) as file:

            notes = file.read().strip()

        if not notes:
            return "You do not have any saved notes, Sir."

        return notes

    except Exception as e:

        logging.exception(e)

        return "I could not read the notes, Sir."

def listen():

    if not MICROPHONE_AVAILABLE:

        return None

    try:

        with microphone as source:

            update_status("Listening...")

            recognizer.adjust_for_ambient_noise(
                source,
                duration=0.5
            )

            audio = recognizer.listen(
                source,
                timeout=5,
                phrase_time_limit=10
            )

        update_status("Recognizing...")

        text = recognizer.recognize_google(
            audio
        )

        return text

    except sr.WaitTimeoutError:

        return None

    except sr.UnknownValueError:

        return None

    except sr.RequestError:

        return (
            "Speech recognition requires an "
            "internet connection."
        )

    except Exception as e:

        logging.exception(e)

        return None

def process_command(command):

    if not command:
        return

    command = command.strip()

    if not command:
        return

    display_user(command)

    lower = command.lower()

    # --------------------------------------------------------
    # EXIT
    # --------------------------------------------------------

    if lower in [
        "exit",
        "quit",
        "goodbye",
        "bye",
        "shutdown jarvis"
    ]:

        answer = "Goodbye, Sir."

        display_jarvis(answer)
        speak(answer)

        root.after(
            1800,
            root.destroy
        )

        return

    # --------------------------------------------------------
    # TIME
    # --------------------------------------------------------

    if lower in [
        "time",
        "what time is it",
        "tell me the time"
    ]:

        answer = tell_time()

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # DATE
    # --------------------------------------------------------

    if lower in [
        "date",
        "what is the date",
        "today's date",
        "tell me the date"
    ]:

        answer = tell_date()

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # GOOGLE
    # --------------------------------------------------------

    if lower in [
        "google",
        "open google"
    ]:

        answer = open_google()

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # YOUTUBE
    # --------------------------------------------------------

    if lower in [
        "youtube",
        "open youtube"
    ]:

        answer = open_youtube()

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # SEARCH
    # --------------------------------------------------------

    if lower.startswith("search "):

        query = command[7:].strip()

        if query:

            answer = google_search(query)

            display_jarvis(answer)
            speak(answer)

        return

    # --------------------------------------------------------
    # YOUTUBE SEARCH / PLAY
    # --------------------------------------------------------

    if lower.startswith("play "):

        query = command[5:].strip()

        if query:

            answer = youtube_search(query)

            display_jarvis(answer)
            speak(answer)

        return

    if lower.startswith("youtube search "):

        query = command[15:].strip()

        if query:

            answer = youtube_search(query)

            display_jarvis(answer)
            speak(answer)

        return

    # --------------------------------------------------------
    # CALCULATOR
    # --------------------------------------------------------

    if lower.startswith("calculate "):

        expression = command[10:].strip()

        try:

            result = calculate(expression)

            answer = f"The answer is {result}, Sir."

        except Exception:

            answer = (
                "I could not calculate that expression, Sir."
            )

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # APPS
    # --------------------------------------------------------

    if lower in [
        "open notepad",
        "notepad"
    ]:

        answer = open_notepad()

        display_jarvis(answer)
        speak(answer)

        return

    if lower in [
        "open calculator",
        "calculator"
    ]:

        answer = open_calculator()

        display_jarvis(answer)
        speak(answer)

        return

    if lower in [
        "open file explorer",
        "open explorer",
        "explorer"
    ]:

        answer = open_explorer()

        display_jarvis(answer)
        speak(answer)

        return

    if lower in [
        "open command prompt",
        "open cmd",
        "cmd"
    ]:

        answer = open_cmd()

        display_jarvis(answer)
        speak(answer)

        return

    if lower in [
        "open powershell",
        "powershell"
    ]:

        answer = open_powershell()

        display_jarvis(answer)
        speak(answer)

        return

    if lower in [
        "open paint",
        "paint"
    ]:

        answer = open_paint()

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # SCREENSHOT
    # --------------------------------------------------------

    if lower in [
        "screenshot",
        "take screenshot",
        "take a screenshot"
    ]:

        answer = take_screenshot()

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # BATTERY
    # --------------------------------------------------------

    if lower in [
        "battery",
        "battery status",
        "check battery"
    ]:

        answer = battery_status()

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # SYSTEM
    # --------------------------------------------------------

    if lower in [
        "system status",
        "system information",
        "computer status"
    ]:

        answer = system_status()

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # NOTES
    # --------------------------------------------------------

    if lower.startswith("take note "):

        note = command[10:].strip()

        answer = save_note(note)

        display_jarvis(answer)
        speak(answer)

        return

    if lower.startswith("remember "):

        note = command[9:].strip()

        answer = save_note(note)

        display_jarvis(answer)
        speak(answer)

        return

    if lower in [
        "read notes",
        "show notes"
    ]:

        answer = read_notes()

        display_jarvis(answer)

        # Don't speak a huge notes file.
        if len(answer) < 500:
            speak(answer)

        return

    # --------------------------------------------------------
    # CLEAR MEMORY
    # --------------------------------------------------------

    if lower == "clear memory":

        clear_memory()

        answer = "Conversation memory has been cleared, Sir."

        display_jarvis(answer)
        speak(answer)

        return

    # --------------------------------------------------------
    # HELP
    # --------------------------------------------------------

    if lower in [
        "help",
        "commands",
        "show commands"
    ]:

        answer = (
            "Available commands:\n\n"
            "Time\n"
            "Date\n"
            "Open Google\n"
            "Open YouTube\n"
            "Search [query]\n"
            "Play [song]\n"
            "Calculate [expression]\n"
            "Open Notepad\n"
            "Open Calculator\n"
            "Open Explorer\n"
            "Open CMD\n"
            "Open PowerShell\n"
            "Open Paint\n"
            "Take Screenshot\n"
            "Battery Status\n"
            "System Status\n"
            "Take Note [text]\n"
            "Read Notes\n"
            "Clear Memory\n"
            "Exit"
        )

        display_jarvis(answer)
        speak(
            "I have displayed my available commands, Sir."
        )

        return

    # --------------------------------------------------------
    # AI FALLBACK
    # --------------------------------------------------------

    update_status("Thinking...")

    def ai_thread():

        answer = ask_ai(command)

        root.after(
            0,
            lambda: display_jarvis(answer)
        )

        speak(answer)

        root.after(
            0,
            lambda: update_status("Online")
        )

    threading.Thread(
        target=ai_thread,
        daemon=True
    ).start()

    chat_box.insert(

def update_status(text):

    root.after(
        0,
        lambda: status_label.config(
            text="● " + text
        )
    )


def send_command(event=None):

    command = input_box.get().strip()

    if not command:
        return

    input_box.delete(
        0,
        tk.END
    )

    threading.Thread(
        target=process_command,
        args=(command,),
        daemon=True
    ).start()


def voice_command():

    if not MICROPHONE_AVAILABLE:

        display_jarvis(
            "Microphone is not available, Sir."
        )

        return

    def voice_thread():

        text = listen()

        if text:

            root.after(
                0,
                lambda: process_command(text)
            )

        else:

            root.after(
                0,
                lambda: update_status("Online")
            )

    threading.Thread(
        target=voice_thread,
        daemon=True
    ).start()

# WAKE WORD MODE
# ============================================================

wake_mode = False
wake_thread_running = False


def toggle_wake_mode():

    global wake_mode

    wake_mode = not wake_mode

    if wake_mode:

        wake_button.config(
            text="WAKE MODE: ON"
        )

        update_status("Wake mode active")

        start_wake_listener()

    else:

        wake_button.config(
            text="WAKE MODE: OFF"
        )

        update_status("Online")


def start_wake_listener():

    global wake_thread_running

    if wake_thread_running:
        return

    wake_thread_running = True

    threading.Thread(
        target=wake_listener,
        daemon=True
    ).start()


def wake_listener():

    global wake_thread_running

    while wake_mode:

        try:

            with microphone as source:

                update_status("Waiting for Hey Jarvis...")

                audio = recognizer.listen(
                    source,
                    timeout=5,
                    phrase_time_limit=5
                )

            text = recognizer.recognize_google(
                audio
            ).lower()

            logging.info(
                f"Wake listener heard: {text}"
            )

            if (
                "hey jarvis" in text
                or "hey jarvis" == text
            ):

                root.after(
                    0,
                    lambda: display_jarvis(
                        "Yes, Sir?"
                    )
                )

                speak("Yes, Sir.")

                time.sleep(0.5)

                command = listen()

                if command:

                    root.after(
                        0,
                        lambda c=command:
                        process_command(c)
                    )

        except Exception:
            pass

    wake_thread_running = False

    update_status("Online")
